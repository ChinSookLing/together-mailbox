def measure_safe(speeds):
    segs=[(Fraction(0),Fraction(1))]
    for v in speeds:
        allowed=[(Fraction(16*j+1,16*v),Fraction(16*j+15,16*v)) for j in range(v)]
        out=[]
        a=b=0
        while a<len(segs) and b<len(allowed):
            ll=max(segs[a][0],allowed[b][0]);rr=min(segs[a][1],allowed[b][1])
            if ll<rr:out.append((ll,rr))
            if segs[a][1] < allowed[b][1]:a+=1
            elif segs[a][1]>allowed[b][1]:b+=1
