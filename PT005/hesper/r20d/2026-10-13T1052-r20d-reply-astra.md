2. 方法与 Soundness
PROOF：
令 N = 16p。每个子代为 w_i' = w_i + b_i N, b_i ∈ {0,1}。由于 N 是偶数，子代的奇偶性与父代一致。因此，只要父代恰有两条奇速度，所有子代都继续满足 odd-coordinate screen。
对每一个奇数时间 T ∈ {1, 3, …, N−1}, 检查是否存在坐标 i 满足 16·min(T w_i' mod 2N, 2N − (T w_i' mod 2N)) < 2N。
PROOF：每个偶数时间 T = 2S 的覆盖性质由父代在 level N 的覆盖性质保证，因此只需检查奇数时间。这正是附件定义的下一层 improper 判定。
本轮 C++ 程序使用精确整数和位集并集，对每种 b 组合完整判定，没有概率测试或可能丢行的启发式剪枝。
核心代码：
C++
constexpr int p = 47;
constexpr int K = 15;
constexpr int N = 16*p;
constexpr int NN = 2*N;
// Bitset covers all odd times T < N.
constexpr int nt = N/2;
constexpr int NW = (nt+63)/64;
using Bits = array<uint64_t, NW>;
Bits coverOdd(int v) {
  Bits result{};
  for (int j = 0; j < nt; ++j) {
    int T = 2*j + 1;
    int r = (1LL * T * v) % NN;
    if (16 * min(r, NN - r) < NN)
      result[j/64] |= 1ULL << (j%64);
  }
  return result;
}
// w is an independently verified improper parent at level 16p.
int countChildren(const array<int, K>& w) {
  Bits opts[K][2];
  for (int i = 0; i < K; ++i) {
    opts[i][0] = coverOdd(w[i]);
    opts[i][1] = coverOdd(w[i] + N);
  }
  int odd = 0;
  for (int v : w) odd += v & 1;
  if (odd < 2) return 0;
  int count = 0;
  for (int mask = 0; mask < (1 << K); ++mask) {
    Bits covered{};
    for (int i = 0; i < K; ++i) {
      const auto& c = opts[i][(mask >> i) & 1];
      for (int j = 0; j < NW; ++j)
        covered[j] |= c[j];
    }
    bool full = true;
    for (int j = 0; j < nt; ++j) {
      if (!(covered[j/64] & (1ULL << (j%64)))) {
        full = false;
        break;
      }
    }
    if (full) ++count;
  }
  return count;
}
