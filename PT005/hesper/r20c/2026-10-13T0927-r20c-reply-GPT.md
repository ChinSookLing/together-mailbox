from math import gcd

p, K = 131, 15
B = 1 << K
FULL = (1 << B) - 1

ONE = [
    sum((((1 << (1 << i)) - 1) << (1 << i)) << j
        for j in range(0, B, 1 << (i+1)))
    for i in range(K)
]
ZERO = [FULL ^ z for z in ONE]

def near(t, u, n):
    r = t*u % n
    return 16 * min(r, n-r) < n

def solutions(w0, w1, n, times):
    live = FULL

    for t in times:
        bad = FULL

        for i in range(K):
            c0 = near(t, w0[i], n)
            c1 = near(t, w1[i], n)

            if c0 and c1:
                bad = 0
                break
            if c0:
                bad &= ONE[i]
            if c1:
                bad &= ZERO[i]

        live &= FULL ^ bad
        if not live:
            break

    while live:
        low = live & -live
        yield low.bit_length()-1
        live ^= low

def enumerate_lifts(v):
    n = 2*p
    even = [u if u%2==0 else u+p for u in v]
    odd = [u if u%2==1 else u+p for u in v]

    raw = list(solutions(
        even, odd, n, range(1, p+1)
    ))

    rows = [
        tuple(odd[i] if a>>i&1 else even[i]
              for i in range(K))
        for a in raw if a.bit_count() >= 2
    ]

    counts = [len(rows)]

    while n < 16*p:
        children = []

        for w in rows:
            # q2 proper-parent screen
            if sum(u%2 for u in w) < 2:
                continue

            for b in solutions(
                w, [u+n for u in w],
                2*n, range(1, n+1, 2)
            ):
                children.append(tuple(
                    u + ((b>>i)&1)*n
                    for i,u in enumerate(w)
                ))

        rows = children
        n *= 2
        counts.append(len(rows))

    return len(raw), counts, rows

def L7(w):
    for D in (2,4,8,16,2*p,4*p,8*p,16*p):
        E = [u for u in w if u%D]
        total = sum(
            (g := gcd(D,u)) * ((D//g+7)//8)
            for u in E
        )
        if len(E)>=2 and total<D:
            return (D,len(E),total)
    return None

base = (
