# Job k11-p383a (Opus letter ecf0ff2): second-machine run of gate p = 383 part (a), office kit unchanged.
# Build once with jobs/kit/setup_k15.sh (sha256 602b6f99...), then kcascade_run.py (sha256 90632d55...) with 16 workers.
# First start: BUDGET stops new jobs at 2026-10-17 05:30 +08. Any start after that runs with no BUDGET.
# Restart-safe: kcascade_run.py skips jobs already in out383/ir.jsonl.
set +x
set -uo pipefail
OUT=/tmp
source "${HOME}/k11-gate/dispatcher/gates/lib.sh"
N=k11-p383a; W="$WORK/$N"; CK="$W/ckpt"; R="$W/out383"; B="$HOME/lr16/code"
install -d "$CK" "$R"
exec >>"$W/run.log" 2>&1
echo "$(date '+%F %T %Z') start $N"
snap() {
  cp -f "$W/run383.time" "$W/hashes.txt" "$W/lscpu.txt" "$CK/" 2>/dev/null
  tail -n 20 "$W/setup.log" > "$CK/setup.log.tail" 2>/dev/null
  tail -n 20 "$W/run383.log" > "$CK/run383.log.tail" 2>/dev/null
  cp -f "$R/ir.jsonl" "$R/km1.json" "$CK/" 2>/dev/null
  n=$(cat "$R/ir.jsonl" 2>/dev/null | wc -l)
  printf '{"job":"%s","ir_lines":%s,"of":415,"time":"%s"}\n' "$N" "$n" "$(date '+%F %T %Z')" > "$CK/latest.json.tmp" && mv "$CK/latest.json.tmp" "$CK/latest.json"
}
if [[ ! -f "$W/build.ok" ]]; then
  if ! command -v unzip >/dev/null 2>&1; then
    install -d "$W/bin"; cp "$D/jobs/kit/unzip-shim.py" "$W/bin/unzip"; chmod +x "$W/bin/unzip"; export PATH="$W/bin:$PATH"
    echo "unzip missing: using jobs/kit/unzip-shim.py"
  fi
  bash "$D/jobs/kit/setup_k15.sh" > "$W/setup.log" 2>&1
  grep -q '^SETUP_OK' "$W/setup.log" || { echo "$(date '+%F %T %Z') setup FAILED"; snap; exit 1; }
  (cd "$B" && sha256sum bgk15 cascade_k15p cascade_filter_k_patched.cpp) > "$W/hashes.txt"
  lscpu > "$W/lscpu.txt"
  date '+%F %T %Z' > "$W/build.ok"
fi
cp -f "$D/jobs/kit/kcascade_run.py" "$B/kcascade_run.py"
now=$(date +%s); dl=$(date -d '2026-10-17 05:30:00 +0800' +%s)
( while sleep 300; do snap; done ) & SNAP=$!
cd "$B"
date '+%F %T %Z' | tee -a "$W/run383.time"
if (( now < dl )); then
  echo "BUDGET=$((dl - now))"
  BUDGET=$((dl - now)) python3 kcascade_run.py ./bgk15 ./cascade_k15p 383 16 "$R" 2>&1 | tee -a "$W/run383.log"
else
  echo "no BUDGET"
  python3 kcascade_run.py ./bgk15 ./cascade_k15p 383 16 "$R" 2>&1 | tee -a "$W/run383.log"

rc=${PIPESTATUS[0]}
date '+%F %T %Z' | tee -a "$W/run383.time"
kill "$SNAP" 2>/dev/null
n=$(wc -l < "$R/ir.jsonl" 2>/dev/null || echo 0)
echo "$(date '+%F %T %Z') kcascade rc=$rc ir_lines=$n"
snap
if (( rc == 0 && n >= 415 )); then date '+%F %T %Z' > "$STATE/$N.done"; echo "DONE $N"; fi

## dispatcher/jobs/kit/unzip-shim.py
```python
#!/usr/bin/env python3
# Used only if the unzip command is missing: does what setup_k15.sh asks, "unzip -q ZIP 'code/*'".
import sys, zipfile, fnmatch
a = [x for x in sys.argv[1:] if x != "-q"]
z = zipfile.ZipFile(a[0]); pats = a[1:] or ["*"]
z.extractall(".", [n for n in z.namelist() if any(fnmatch.fnmatch(n, p) for p in pats)])
```
