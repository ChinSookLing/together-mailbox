+# Prints the six values, one "key value" per line, in a fixed order, or refuses.
+# Each key exactly once; no other line except blank lines and lines starting with #.
+parse_ticket() {
+  local file="$1" key val rest seen=" "
+  local keys="tree_commit tree_prefix read_record_commit read_record_path read_record_sha256 new_pin"
+  while IFS= read -r line || [[ -n "$line" ]]; do
+    line="${line%$'\r'}"
+    [[ -z "$line" || "$line" == \#* ]] && continue
+    [[ "$line" =~ ^([a-z_0-9]+)\ ([^ ]+)$ ]] || refuse "ticket line is not 'key value'"
+    key="${BASH_REMATCH[1]}"; val="${BASH_REMATCH[2]}"
+    case " $keys " in
+      *" $key "*) ;;
+      *) refuse "ticket has an unknown key: $key" ;;
+    esac
+    [[ "$seen" != *" $key "* ]] || refuse "ticket names $key twice"
+    seen="${seen}${key} "
+    printf -v "T_${key}" '%s' "$val"
+  done < "$file"
+  for key in $keys; do
+    [[ "$seen" == *" $key "* ]] || refuse "ticket has no $key"
+  done
+  for key in $keys; do
+    rest="T_${key}"
+    printf '%s %s\n' "$key" "${!rest}"
+  done
+}
