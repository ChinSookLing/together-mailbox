def next_improper(w, p, k):
    N = (2**k) * p
    M = 2 * N
    odd_times = range(1, N, 2)


    options = []
    for a in w:
        pair = []
        for value in (a, a + N):
            mask = 0
            for i, t in enumerate(odd_times):
                r = (t * value) % M
                if 16 * min(r, M-r) < M:
                    mask |= 1 << i
            pair.append(mask)
        options.append(pair)


    FULL = (1 << (N//2)) - 1
    suffix = [0] * (len(w)+1)


    for i in range(len(w)-1, -1, -1):
        suffix[i] = (
            suffix[i+1]
            | options[i][0]
            | options[i][1]
        )


    out = []


    def visit(i, covered_mask, vals):
        if (covered_mask | suffix[i]) != FULL:
            return


        if i == len(w):
            out.append(tuple(vals))
