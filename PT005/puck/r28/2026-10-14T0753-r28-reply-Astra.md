from collections import Counter


W = [3,125,419,541,269,275,397,147]


def tile(k,b,w):
    N = 17 * 2**k
    return [
        (t if t%17==b else -t) % (2**k)
        for t in range(1,N//2+1,2)
        if t%17 in (b,17-b)
        and 16*min(t*w%N,(-t*w)%N)<N
    ]


A = [tile(6,1,w) for w in W]
B = [tile(7,2,w) for w in W]
