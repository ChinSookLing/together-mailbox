### analyze.py
import json, glob
from collections import defaultdict, Counter
from predict import predict
rows=[]
for f in sorted(glob.glob('rows_p*.json'),key=lambda x:int(x.split('_p')[1][:-5])):
    rows+=json.load(open(f))
per=defaultdict(lambda:[0,0,0,0,0,0])
kf=defaultdict(set); kp=defaultdict(set)
for r in rows:
    key=tuple(tuple(x) for x in r[2]); pr,_=predict(key)
    s=per[r[0]]; s[0]+=1; s[1]+=r[3]
    if len(r)>7: s[2]+=r[7]; s[3]+=r[5]; s[4]+=(r[5] and r[3])
    s[5]+= (pr!=r[3])
    kf[key].add(r[3]); kp[key].add(r[0])
print('p tested doublable 2cover_mod64 mixed_tested mixed_doublable predictor_mismatch')
for p in sorted(per): print(p,*per[p])
print('total rows',len(rows),'distinct reduced keys',len(kf),'keys with conflicting flags',sum(len(v)>1 for v in kf.values()))
multi=[k for k in kf if len(kp[k])>=2]
print('keys seen at >=2 primes',len(multi),'of which doublable',sum(True in kf[k] for k in multi))
dk=[k for k in kf if True in kf[k]]
print('distinct doublable keys',len(dk),'primes-per-doublable-key hist',sorted(Counter(len(kp[k]) for k in dk).items()))
print('doublable keys with an odd c:',sum(any(c%2 for c,_,_ in k) for k in dk))
