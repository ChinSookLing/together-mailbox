from math import gcd

def improper(w,p,k):
    N=(1<<k)*p
    if sum(x%2 for x in w)<2:
        return False
    for T in range(1,N//2+1):
        covered=False
        for v in w:
            r=(T*v)%N
            if 16*min(r,N-r)<N:
                covered=True
                break
        if not covered:
            return False
    return True

def children(w,p,k):
    N=(1<<k)*p
    out=[]
    for mask in range(1<<len(w)):
        c=[w[i]+(((mask>>i)&1)*N)
           for i in range(len(w))]
        if sum(x%2 for x in c)<2:
            continue
        if improper(c,p,k+1):
            out.append(c)
    return out

def frozen(w,k):
    return sum(
        ((x & -x).bit_length()-1)<k
        for x in w
    )
