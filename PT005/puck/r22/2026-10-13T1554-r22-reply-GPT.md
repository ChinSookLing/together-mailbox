#include <bits/stdc++.h>
using namespace std;

using W = array<int,15>;
using U = uint64_t;

vector<int> L7(const W &w, int N) {
    vector<int> yes;

    for (int D=2; D<=N; D+=2) {
        if (N%D) continue;

        int e=0, sum=0;

        for (int x:w) {
            if (x%D) {
                int g=gcd(x,D);
                ++e;
                sum += g*((D/g+7)/8);
            }
        }

        if (e>=2 && sum<D)
            yes.push_back(D);
    }
    return yes;
}

bool improper(const W &w, int N) {
    int odd=0;
    for (int x:w) odd+=x%2;
    if (odd<2) return false;

    for (int t=1; t<=N/2; t++) {
        bool covered=false;

        for (int x:w) {
            int r=1LL*t*x%N;
            if (16LL*min(r,N-r)<N) {
                covered=true;
                break;
            }
        }
        if (!covered) return false;
    }
    return true;
}

struct Result {
    long long hits=0, badcount=0;
    map<int,long long> passingD;
    vector<W> bad;
};

Result lift(const W &base, int N,
            bool saveBad=true) {
    int M=2*N, T=M/2;
    int q=(T+63)/64;

    vector<array<vector<U>,2>> b(15);

    for (int i=0;i<15;i++)
        for (int v=0;v<2;v++) {
            b[i][v].assign(q,0);
            int x=base[i]+v*N;

            for (int t=1;t<=T;t++) {
                int r=1LL*t*x%M;

                if (16LL*min(r,M-r)<M)
                    b[i][v][(t-1)/64] |=
                        U(1)<<((t-1)%64);
            }
        }

    vector<vector<U>> bits(
        16,vector<U>(q));

    U tail=T%64?
        ((U(1)<<(T%64))-1):~U(0);

    W cur{};
    Result res;

    auto rec=[&](auto &&self,int i)->void {
        if (i==15) {
            for (int j=0;j<q;j++)
                if (bits[15][j] !=
                    (j==q-1?tail:~U(0)))
                    return;

            int odd=0;
            for (int x:cur) odd+=x%2;
            if (odd<2) return;

            res.hits++;

            auto Ds=L7(cur,M);

            if (Ds.empty()) {
                res.badcount++;
                if (saveBad)
                    res.bad.push_back(cur);
            } else {
                for (int d:Ds)
                    res.passingD[d]++;
            }
            return;
        }

        for (int v=0;v<2;v++) {
            cur[i]=base[i]+v*N;

            for (int j=0;j<q;j++)
                bits[i+1][j] =
                    bits[i][j] | b[i][v][j];

            self(self,i+1);
        }
    };

    rec(rec,0);
    return res;
}
