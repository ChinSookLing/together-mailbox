以下为实际 C++ 程序中的核心递归：

```

int closure(
    const V& w,
    int depth,
    vector<V>& route
) {
    // A valid L7 certificate closes
    // this entire descendant subtree.
    if (l7(w, depth))
        return depth;

    // Kill condition: L7 fails at 256p
    // and an improper child exists at 512p.
    if (depth == 8) {
        Level next(*this, w, (1 << depth) * p);
        V first{};
        auto save = [&](const V& v) {
            first = v;
        };

        bool hasChild = next.each(save, true);

        if (hasChild) {
            printVec("fail_at_256p", w);
            printVec("improper_child_512p", first);
            printL7(w, depth);
            throw runtime_error("KILL");
        }

        return 9;
    }

    Level next(*this, w, (1 << depth) * p);
    int deepest = depth + 1;

    auto check = [&](const V& child) {
        route.push_back(child);
        int d = closure(child, depth + 1, route);
        route.pop_back();
        deepest = max(deepest, d);
    };

    // Enumerate every improper child.
    // The Level enumerator implements
    // odd-time coverage and the >=2 odd screen.
    next.each(check, false);
    return deepest;
}

```
