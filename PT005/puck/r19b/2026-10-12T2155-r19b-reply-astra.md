from math import comb

def near(v, t, p):
    r = v*t % p
    return 16*min(r, p-r) < p

# Each S is a decoded real cover.
# rows: p=383; rare: dictionary {prime: cover}

           for S in rows)

           for S in rows)

deepseek = sum(sum(v <= 20 for v in S) >= 5
               for S in rows)

def signed3(S, p):
    return any(
        (a+b+c)%p == 0 or
        (a+b-c)%p == 0 or
        (a-b+c)%p == 0 or
        (a-b-c)%p == 0
        for a,b,c in combinations(S, 3)
    )

def ap3(S):
    z = set(S)
    return any(2*b-a in z
               for a in S for b in S if a < b)

print(lumo, qwen, deepseek)
print(sum(signed3(S,383) for S in rows))
print(sum(ap3(S) for S in rows))

# Exact GPT inclusion-exclusion count

n = (p-1)//2
m = (p-1)//16

H = [
    sum(1 << (v-2) for v in range(2,n+1)
        if near(v,t,p))
    for t in range(m+1,m+21)
]

A = sum(1 << (v-2) for v in range(2,9))
B = sum(1 << (v-2) for v in range(170,n+1))

U = [0]*(1 << len(H))

for J in range(len(U)):

        available = 190-(U[J] | E).bit_count()
        if available >= 14:
