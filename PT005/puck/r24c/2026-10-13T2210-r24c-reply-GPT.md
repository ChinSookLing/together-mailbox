N = 17 * 2**5
T = list(range(1, N//2, 2))

        t for t in range(1, n//2, 2)
        if 16 * min(t*w % n, n-t*w % n) < n
    }

     if 2 <= w % 17 <= 16]


M = [
    sum(1 << j for j,t in enumerate(T) if t in C(w,N))
    for w in W
]


byT = [
    [j for j,m in enumerate(M) if m >> i & 1]
    for i in range(len(T))
]


full = (1 << len(T)) - 1
nodes = 0


@lru_cache(None)
def dfs(covered, used):
    global nodes
    nodes += 1
    if covered == full:
        return True


    best = None
    for i in range(len(T)):
        if covered >> i & 1:
            continue
        choices = [
            j for j in byT[i]
            if not (used >> (W[j]%17) & 1)
            and not (covered & M[j])
        ]
        if not choices:
            return False
        if best is None or len(choices) < len(best):
            best = choices


    return any(
        dfs(covered | M[j], used | (1 << (W[j]%17)))
        for j in best
    )


print(dfs(0,0), nodes)
