function census(k) {
  const p = 17, N = p * 2**k;
  const m = N / 4, B = Math.ceil(m / 32);
  const a = [];

  for (let w = 1; w < N/2; w++) {
    if (w % p === 0) continue;

    let z = w, v = 0;
    while (!(z & 1)) { v++; z >>= 1; }
    if (v > k-5) continue;

    const bits = new Uint32Array(B);
    let count = 0;

    for (let t = 1; t < N/2; t += 2) {
      const r = t*w % N;
      if (16*Math.min(r,N-r) < N) {
        const j = (t-1)/2;
        bits[j>>>5] |= 1<<(j&31);
        count++;
      }
    }
    if (count !== m/8) throw Error("bad count");
    a.push({w, r:w%p, bits});
  }

  const adj = a.map(() => []);
  for (let i=0; i<a.length; i++)
    for (let j=i+1; j<a.length; j++) {
      let disjoint = true;
      for (let b=0; b<B; b++)
        if (a[i].bits[b] & a[j].bits[b])
          { disjoint=false; break; }
      if (disjoint) adj[i].push(j);
    }

  let tilings=0, maxClasses=0, legal=0;

  function dfs(ch,cand) {
    if (ch.length===8) {
      tilings++;
      const rs = new Set(ch.flatMap(i =>
        [a[i].r,17-a[i].r]));
      maxClasses=Math.max(maxClasses,rs.size);
      if (rs.size>=8) legal++;
      return;
    }
    const need=8-ch.length;
    for (let h=0; h<=cand.length-need; h++) {
      const i=cand[h], next=[];
      let t=h+1, j=0, nb=adj[i];
      while (t<cand.length && j<nb.length) {
        if (cand[t]===nb[j]) {
          next.push(cand[t]); t++; j++;
        } else if (cand[t]<nb[j]) t++;
        else j++;
      }
      if (next.length>=need-1)
        dfs([...ch,i],next);
    }
  }

  dfs([],a.map((_,i)=>i));
  return {k,m,perSet:m/8,tilings,maxClasses,legal};
}
