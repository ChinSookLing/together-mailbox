from collections import defaultdict

N = 544

def masks(w,b):
    a=w%32
    A=pow(a,-1,32)
    z=b*w%17
    c=z if z%2 else z-17
    s=1 if c>0 else -1
    e=2*c-17*s

    L=[c,c-2*s]
    U=([e-4,e-2,e,e+2] if e>0
       else [e-2,e,e+2,e+4])

    def mask(xs):
        return sum(1<<((A*x%32)//2) for x in xs)

    return mask(L),mask(U)

def partitions(b,h):
    V=[]
    for w in range(1,272,2):
        if w%17 in (0,1,16):
            continue
        if w%8 not in (h,8-h):
            continue
        L,U=masks(w,b)
        assert L.bit_count()==2
        assert U.bit_count()==4
        V.append((w,L,U))

    adj=[[] for _ in range(16)]
    for v in V:
        for x in range(16):
            if v[2]>>x&1:
                adj[x].append(v)

    result=defaultdict(list)

    def dfs(U,L,W):
        if U==65535:
            assert len(W)==4
            result[L].append(W)
            return

               if not (U>>x)&1)

        for w,l,u in adj[x]:
            if not (U&u or L&l):
                dfs(U|u,L|l,W+(w,))

    dfs(0,0,())
    return result
