def target(b,k):
    M=17*(1<<k)
    return [t for t in range(1,M//2+1,2)
            if t%17 in (b%17,(-b)%17)]

def dist(x):

def partitions(b,cls):
    T=target(b,5)
    U=target(2*b,6)
    assert len(T)==16 and len(U)==32

    W=[w for w in range(1,N//2+1,2)
       if w%17 not in (0,1,16)
       and w%8 in ((1,7) if cls==1 else (3,5))]

    base={
      w:sum(1<<j for j,t in enumerate(T)
            if 16*dist(t*w)<N)
      for w in W}

    lifted={
      w:sum(1<<j for j,t in enumerate(U)
            if 8*dist(t*w)<N)
      for w in W}

    assert all(x.bit_count()==2 for x in base.values())
    assert all(x.bit_count()==8 for x in lifted.values())

    at=[[w for w in W if lifted[w]&(1<<j)]
        for j in range(32)]

    full=(1<<32)-1
    result=[]

    def visit(mask,foot,chosen):
        if mask==full:
            assert len(chosen)==4
            result.append((tuple(sorted(chosen)),foot))
            return

        if len(chosen)==4:
            return

        j=next(j for j in range(32)
               if not mask&(1<<j))

        for w in at[j]:
            if not(mask&lifted[w]) and not(foot&base[w]):
                visit(mask|lifted[w],
                      foot|base[w],
                      chosen+[w])

    visit(0,0,[])
    assert len(result)==len(set(result))
    return result
