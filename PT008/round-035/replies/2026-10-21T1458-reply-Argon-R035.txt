from fractions import Fraction as F

def dist(x):
    r = x - int(x)
    if r < 0: r += 1
    return min(r, 1 - r)

def min_dist(v, t):
    return min(dist(vi * t) for vi in v)

# 检验紧实例 v* = (1, 2, 3) 及其单侧敌意扰动 v(eps) = (1 - eps, 2, 3)
eps = F(1, 1000000)
v_star = (F(1), F(2), F(3))
v_pert = (F(1) - eps, F(2), F(3))

# 分支 1: t1(v) = 1 / (4 * v1)
# 分支 2: t2(v) = 3 / (v1 + v3)
for name, v in [("v* = (1,2,3)", v_star), ("v(eps) = (1-10^-6, 2, 3)", v_pert)]:
    t1 = F(1) / (4 * v[0])
    t2 = F(3) / (v[0] + v[2])
    d1 = min_dist(v, t1)
    d2 = min_dist(v, t2)
    print(f"{name}:")
    print(f"  Branch 1 (t1 = {t1}): min_dist = {d1} >= 1/4? {d1 >= F(1,4)}")
    print(f"  Branch 2 (t2 = {t2}): min_dist = {d2} >= 1/4? {d2 >= F(1,4)}")
